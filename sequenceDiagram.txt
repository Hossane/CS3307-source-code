@startuml
skinparam sequenceMessageAlign center
skinparam responseMessageBelowArrow true
title UC6 — Approve Access Request (main + audit-failure)

actor Approver
participant ":AuthService" as AS
participant ":RequestService" as RS
participant ":ApprovalPolicy" as AP
participant ":PermissionService" as PS
participant ":AuditService" as AudS
participant ":Clock" as Clock
participant ":Notifier" as Notif
participant ":accessRequest" as Req
participant ":permission" as Perm
participant ":auditLog" as Log

Approver -> AS : log in
activate AS
AS --> Approver : session created
deactivate AS

Approver -> RS : open pending requests
activate RS
RS --> Approver : list of requests
deactivate RS

Approver -> RS : approve request
activate RS

RS -> Req : check not self-approval
activate Req
Req --> RS : not self
deactivate Req

RS -> AP : evaluate request against policy
activate AP
AP --> RS : allowed
deactivate AP

RS -> PS : create time-limited permission
activate PS
PS -> Clock : current time
activate Clock
Clock --> PS : granted time
deactivate Clock
PS -> Perm : create permission
activate Perm
Perm --> PS : created
deactivate Perm
PS --> RS : permission created
deactivate PS

RS -> Req : mark approved
activate Req
Req --> RS : approved
deactivate Req

RS ->> Notif : notify requester
note right of Notif : asynchronous;\nfailure is non-critical

RS -> AudS : record audit entry
activate AudS

alt audit write succeeds
    AudS -> Clock : current time
    activate Clock
    Clock --> AudS : timestamp
    deactivate Clock
    AudS -> Log : append entry
    activate Log
    Log --> AudS : appended
    deactivate Log
    AudS --> RS : recorded
    RS --> Approver : decision confirmed
else audit write fails
    AudS --> RS : recording failed
    RS -> PS : revoke permission
    activate PS
    PS --> RS : revoked
    deactivate PS
    RS -> Req : revert to submitted
    activate Req
    Req --> RS : reverted
    deactivate Req
    RS --> Approver : decision rolled back
end

deactivate AudS
deactivate RS

@enduml
